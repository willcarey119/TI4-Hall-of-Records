## Type definitions (referenced by tasks below)

```ts
// New types added to buildAgendaCrossGame.ts and re-exported from aggregator/index.ts

/** One game's resolution of one agenda, enriched with cross-game context. */
export interface AgendaAppearance {
  gameId: string;
  /** First-action timestamp for the game; used for label formatting + sort. */
  gameDate: number;
  /** Faction id of the game's winner, or null if no winner. */
  gameWinner: string | null;
  round: number;
  /** 1 or 2 — which agenda within the round (mirrors AgendaDisplayEntry). */
  indexInRound: 1 | 2;
  outcome: string;
  /** True for binary 'For' OR for any successfully-resolved elect. */
  passed: boolean;
  totalFor: number;
  totalAgainst: number;
  /** Set when this is an elect-type agenda. */
  electedFaction?: string;
  votes: AgendaVote[];
  riders: AgendaRider[];
  /** Net VP attributable to this resolution, by faction. Source = vpEvents
   *  with source === 'agenda' AND timestamp === resolution.timestamp. */
  vpDeltaByFaction: Record<string, number>;
}

/** Cross-game stats for one unique agenda name. Extends AgendaFreqStat. */
export interface AgendaCrossGameStat extends AgendaFreqStat {
  /** All appearances of this agenda, sorted by gameDate ascending. */
  appearances: AgendaAppearance[];
  /** Net VP this agenda has swung across all its appearances, by faction. */
  vpDeltaByFaction: Record<string, number>;
  /** Histogram: round number → count of appearances in that round. */
  roundDistribution: Record<number, number>;
}

/** Per-faction voting pattern across one agenda. */
export interface FactionAgendaVote {
  factionId: string;
  /** Times this faction voted For on this agenda. */
  forCount: number;
  /** Times Against. */
  againstCount: number;
  /** Times voted for an elect target (for elect-type agendas). */
  electCount: number;
  /** Times abstained (appeared in game where agenda resolved but cast no vote). */
  abstainCount: number;
}

/** Cross-faction voting pattern for one agenda. Used by FactionVotingPanel. */
export interface AgendaFactionVotingSummary {
  agenda: string;
  factions: FactionAgendaVote[];
}

export interface AgendaCrossGameSummary {
  agendas: AgendaCrossGameStat[];
  totalResolutions: number;
  overallPassRate: number | null;
  gamesAnalyzed: number;
  /** Per-agenda faction voting matrix. Computed once, indexable by agenda name. */
  factionVotingByAgenda: Record<string, AgendaFactionVotingSummary>;
}
```
